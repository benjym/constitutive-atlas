cmake_minimum_required(VERSION 3.20)
project(AtlasTraceContext LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ATLAS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${ATLAS_ROOT}/cmake/AtlasScalarFunctions.cmake")
set(STUDY "${ATLAS_ROOT}/catalogue/investigations/trace-generated-context-v1")
add_executable(atlas_trace_context EXCLUDE_FROM_ALL "${STUDY}/probe.cpp")
target_include_directories(atlas_trace_context PRIVATE "${ATLAS_ROOT}/cpp/include" "${STUDY}/generated")
target_compile_options(atlas_trace_context PRIVATE -Wall -Wextra -Wno-unused-variable -fno-fast-math -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(atlas_trace_context PRIVATE -fexceptions)
 set_target_properties(atlas_trace_context PROPERTIES OUTPUT_NAME atlas-trace-context SUFFIX ".js")
 target_link_options(atlas_trace_context PRIVATE -fexceptions "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=node" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-sALLOW_MEMORY_GROWTH=1" "-sSTACK_SIZE=1048576" "-sFILESYSTEM=0" "--no-entry")
endif()
