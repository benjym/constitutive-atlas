void number(std::ostream&o,double x){if(std::isfinite(x))o<<std::setprecision(17)<<x;else o<<"null";}
void array(std::ostream&o,const State&v){o<<'[';for(size_t i=0;i<v.size();++i){if(i)o<<',';number(o,v[i]);}o<<']';}

void diagnostics(std::ostream&o,const Diagnostics&d){o<<"{\"substeps\":"<<d.substeps<<",\"partition_retries\":"<<d.partition_retries<<",\"increment_fractions\":";array(o,d.increment_fractions);o<<",\"accepted_increments\":[";for(size_t i=0;i<d.accepted_increments.size();++i){if(i)o<<',';array(o,d.accepted_increments[i]);}o<<"]}";}
void advance_json(std::ostream&o,const Advance&a){o<<"{\"ok\":"<<(a.ok?"true":"false")<<",\"state\":";array(o,a.state);o<<",\"diagnostics\":";diagnostics(o,a.diagnostics);o<<'}';}
Advance synthetic(const Sym<double>&t,const State&fractions={1.},int retry=0,double slope=78000000.){
 Advance a;a.ok=true;a.state={0.,slope*t.v[1],slope*t.v[1],0.,0.,0.,.7};a.diagnostics.increment_fractions=fractions;a.diagnostics.partition_retries=retry;a.diagnostics.substeps=int(fractions.size());
 for(double f:fractions){State x;for(double v:t.v)x.push_back(f*v);a.diagnostics.accepted_increments.push_back(x);}return a;
}
void emit(std::ostream&o,const std::string&id,bool passed,const std::string&extra=""){
 if(!all){} // all is an aggregate verdict, never an early-exit selector.
 static bool first=true;if(!first)o<<',';first=false;all=all&&passed;o<<"{\"id\":\""<<id<<"\",\"passed\":"<<(passed?"true":"false")<<extra<<'}';
}
std::string helper_json(const Advance&base,const Sym<double>&total,const Sym<double>&direction,double h,const ControlDerivative&r,int calls){

std::string run(){

 Sym<double>total,direction;direction.v[1]=direction.v[2]=1.;const double h=.0001;auto base=synthetic(total);

  const State untouched=b.state;
  auto eval=[&](const Sym<double>&x){++calls;auto a=synthetic(x);if(mode=="jump-recovery"&&calls==1||mode=="exact-boundary"){a=synthetic(x,{.5,.5});a.state[1]-=15600;a.state[2]-=15600;}
