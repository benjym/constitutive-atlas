cmake_minimum_required(VERSION 3.20)
project(DuncanMaterialAPIV1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EMSCRIPTEN)
  add_executable(atlas_duncan_api frozen/cpp/main.cpp inspector.cpp)
  target_compile_options(atlas_duncan_api PRIVATE -fexceptions)
  set_target_properties(atlas_duncan_api PROPERTIES OUTPUT_NAME "duncan" SUFFIX ".js")
  target_link_options(atlas_duncan_api PRIVATE "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createAtlas" "-sENVIRONMENT=web,worker,node" "-sALLOW_MEMORY_GROWTH=1" "-sMAXIMUM_MEMORY=268435456" "-sSTACK_SIZE=1048576" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-fexceptions" "-sFILESYSTEM=0" "--no-entry")
else()
  add_library(atlas_duncan_api SHARED frozen/cpp/main.cpp inspector.cpp)
endif()
target_include_directories(atlas_duncan_api PRIVATE frozen/cpp/include frozen/generated)
