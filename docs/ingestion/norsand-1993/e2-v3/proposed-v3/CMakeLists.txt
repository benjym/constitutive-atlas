cmake_minimum_required(VERSION 3.20)
project(atlas_controlled_path LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EMSCRIPTEN)
 add_executable(atlas_path_capability generated/path_api.cpp)
 target_link_options(atlas_path_capability PRIVATE --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=node,web,worker -sALLOW_MEMORY_GROWTH=1 "-sEXPORTED_FUNCTIONS=['_path_size','_path_evaluate','_path_advance','_malloc','_free']" "-sEXPORTED_RUNTIME_METHODS=['ccall','cwrap']")
 set_target_properties(atlas_path_capability PROPERTIES SUFFIX ".mjs")
else()
 add_library(atlas_path_capability SHARED generated/path_api.cpp)
endif()
target_include_directories(atlas_path_capability PRIVATE cpp/include generated)
target_compile_options(atlas_path_capability PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
