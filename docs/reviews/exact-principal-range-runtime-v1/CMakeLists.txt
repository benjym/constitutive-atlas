cmake_minimum_required(VERSION 3.20)
project(AtlasExactRange LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_exact_range EXCLUDE_FROM_ALL driver.cpp)
target_include_directories(atlas_exact_range PRIVATE frozen/cpp/include generated)
target_compile_options(atlas_exact_range PRIVATE -Wall -Wextra -Wno-unused-variable -Wno-misleading-indentation -fno-fast-math -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(atlas_exact_range PRIVATE -fexceptions)
 target_link_options(atlas_exact_range PRIVATE -fexceptions "-sENVIRONMENT=node" "-sEXIT_RUNTIME=1" "-sSTACK_SIZE=1048576" "-sALLOW_MEMORY_GROWTH=1")
 set_target_properties(atlas_exact_range PROPERTIES OUTPUT_NAME exact-range SUFFIX ".js")
endif()
