cmake_minimum_required(VERSION 3.20)
project(AtlasTraceOperator LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ATLAS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${ATLAS_ROOT}/cmake/AtlasScalarFunctions.cmake")
add_executable(atlas_trace_probe EXCLUDE_FROM_ALL "${ATLAS_ROOT}/docs/reviews/tensor-trace-implementation-v1/trace_probe.cpp")
target_include_directories(atlas_trace_probe PRIVATE "${ATLAS_ROOT}/cpp/include")
target_compile_definitions(atlas_trace_probe PRIVATE ATLAS_TRACE_INTEGRATED=1)
target_compile_options(atlas_trace_probe PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
if(EMSCRIPTEN)
  set_target_properties(atlas_trace_probe PROPERTIES OUTPUT_NAME atlas-trace SUFFIX ".js")
  target_link_options(atlas_trace_probe PRIVATE "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=node" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-sFILESYSTEM=0" "--no-entry")
endif()
