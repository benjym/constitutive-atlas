cmake_minimum_required(VERSION 3.20)
project(AtlasCenteredMiddle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_centered_middle EXCLUDE_FROM_ALL driver.cpp)
target_include_directories(atlas_centered_middle PRIVATE frozen/cpp/include generated)
target_compile_options(atlas_centered_middle PRIVATE -Wall -Wextra -Wno-unused-variable -Wno-misleading-indentation -fno-fast-math -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(atlas_centered_middle PRIVATE -fexceptions)
 target_link_options(atlas_centered_middle PRIVATE -fexceptions "-sENVIRONMENT=node" "-sEXIT_RUNTIME=1" "-sSTACK_SIZE=1048576" "-sALLOW_MEMORY_GROWTH=1")
 set_target_properties(atlas_centered_middle PROPERTIES OUTPUT_NAME centered-middle SUFFIX ".js")
endif()
