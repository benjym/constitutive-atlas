cmake_minimum_required(VERSION 3.20)
project(AtlasBoundedPartitionQualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_partition_regression EXCLUDE_FROM_ALL regression.cpp)
add_executable(atlas_partition_analytical EXCLUDE_FROM_ALL partition.cpp)
foreach(target atlas_partition_regression atlas_partition_analytical)
 target_include_directories(${target} PRIVATE frozen/cpp/include generated)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-variable -Wno-misleading-indentation -fno-fast-math -ffp-contract=off)
 if(EMSCRIPTEN)
  target_compile_options(${target} PRIVATE -fexceptions)
  target_link_options(${target} PRIVATE -fexceptions "-sENVIRONMENT=node" "-sEXIT_RUNTIME=1" "-sSTACK_SIZE=1048576" "-sALLOW_MEMORY_GROWTH=1")
  set_target_properties(${target} PROPERTIES SUFFIX ".js")
 endif()
endforeach()
