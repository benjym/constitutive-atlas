cmake_minimum_required(VERSION 3.20)
project(AtlasMCCurrentV3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_solver EXCLUDE_FROM_ALL frozen/cpp/main.cpp)
target_include_directories(atlas_solver PRIVATE frozen/cpp/include generated)
target_compile_options(atlas_solver PRIVATE -Wall -Wextra -Wno-unused-variable)
if(EMSCRIPTEN)
 target_compile_options(atlas_solver PRIVATE -fexceptions)
 set_target_properties(atlas_solver PROPERTIES OUTPUT_NAME atlas SUFFIX ".js")
 target_link_options(atlas_solver PRIVATE -fexceptions "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,worker,node" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-sALLOW_MEMORY_GROWTH=1" "-sMAXIMUM_MEMORY=268435456" "-sSTACK_SIZE=1048576" "-sFILESYSTEM=0" "--no-entry")
endif()
