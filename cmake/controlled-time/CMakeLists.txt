cmake_minimum_required(VERSION 3.20)
project(atlas_controlled_time LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ATLAS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ATLAS_ROOT}/cmake/AtlasScalarFunctions.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ATLAS_CONTROLLED_TIME_MODELS "${ATLAS_ROOT}/tests/fixtures/controlled-time/transport.json;${ATLAS_ROOT}/tests/fixtures/controlled-time/capacity-max.json;${ATLAS_ROOT}/tests/fixtures/controlled-time/branch.json;${ATLAS_ROOT}/tests/fixtures/controlled-time/late-domain.json" CACHE STRING "Explicit controlled-time model files")
file(GLOB ATLAS_PYTHON CONFIGURE_DEPENDS "${ATLAS_ROOT}/atlas/*.py")
set(ATLAS_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${ATLAS_GENERATED}/controlled_time_api.cpp" "${ATLAS_GENERATED}/controlled_time_models.hpp" "${ATLAS_GENERATED}/controlled_time_catalogue.json"
 COMMAND "${Python3_EXECUTABLE}" "${ATLAS_ROOT}/tools/generate_controlled_time.py" --output "${ATLAS_GENERATED}" ${ATLAS_CONTROLLED_TIME_MODELS}
 DEPENDS ${ATLAS_CONTROLLED_TIME_MODELS} ${ATLAS_PYTHON} "${ATLAS_ROOT}/schemas/model-v0.3.schema.json" "${ATLAS_ROOT}/tools/generate_controlled_time.py" VERBATIM)
if(EMSCRIPTEN)
 add_executable(atlas_controlled_time_capability "${ATLAS_GENERATED}/controlled_time_api.cpp")
 target_link_options(atlas_controlled_time_capability PRIVATE --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=node,web,worker -sALLOW_MEMORY_GROWTH=1 "-sEXPORTED_FUNCTIONS=['_controlled_time_catalogue_file_hash','_controlled_time_model_hash','_controlled_time_catalogue_hash','_controlled_time_size','_controlled_time_evaluate','_controlled_time_advance','_malloc','_free']" "-sEXPORTED_RUNTIME_METHODS=['ccall','cwrap','HEAPF64','UTF8ToString']" "-sINCOMING_MODULE_JS_API=['wasmBinary','instantiateWasm','locateFile','print','printErr','onAbort']" -fexceptions)
 target_compile_options(atlas_controlled_time_capability PRIVATE -fexceptions)
 set_target_properties(atlas_controlled_time_capability PROPERTIES SUFFIX ".mjs")
else()
 add_library(atlas_controlled_time_capability SHARED "${ATLAS_GENERATED}/controlled_time_api.cpp")
endif()
target_include_directories(atlas_controlled_time_capability PRIVATE "${ATLAS_ROOT}/cpp/include" "${ATLAS_GENERATED}")
target_compile_options(atlas_controlled_time_capability PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
