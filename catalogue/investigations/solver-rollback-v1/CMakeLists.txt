cmake_minimum_required(VERSION 3.20)
project(AtlasSolverRollback LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_solver_rollback EXCLUDE_FROM_ALL probe.cpp)
target_include_directories(atlas_solver_rollback PRIVATE frozen/cpp/include generated)
target_compile_options(atlas_solver_rollback PRIVATE -Wall -Wextra -Wno-unused-variable -Wno-unused-parameter -fno-fast-math -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(atlas_solver_rollback PRIVATE -fexceptions)
 set_target_properties(atlas_solver_rollback PROPERTIES OUTPUT_NAME atlas-solver-rollback SUFFIX ".js")
 target_link_options(atlas_solver_rollback PRIVATE -fexceptions "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=node" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-sALLOW_MEMORY_GROWTH=1" "--no-entry")
endif()
