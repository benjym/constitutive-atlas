cmake_minimum_required(VERSION 3.20)
project(AtlasBarodesyPreview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_barodesy_preview EXCLUDE_FROM_ALL frozen/cpp/main.cpp)
target_include_directories(atlas_barodesy_preview PRIVATE frozen/cpp/include generated)
target_compile_options(atlas_barodesy_preview PRIVATE -Wall -Wextra -Wno-unused-variable -fno-fast-math -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(atlas_barodesy_preview PRIVATE -fexceptions)
 set_target_properties(atlas_barodesy_preview PROPERTIES OUTPUT_NAME atlas-barodesy-preview SUFFIX ".js")
 target_link_options(atlas_barodesy_preview PRIVATE -fexceptions "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,worker,node" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-sALLOW_MEMORY_GROWTH=1" "-sSTACK_SIZE=1048576" "-sFILESYSTEM=0" "--no-entry")
endif()
