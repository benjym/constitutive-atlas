cmake_minimum_required(VERSION 3.20)
project(AtlasDeterminantContexts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ATLAS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(atlas_normalized_determinant_context EXCLUDE_FROM_ALL driver.cpp)
target_include_directories(atlas_normalized_determinant_context PRIVATE "${ATLAS_ROOT}/cpp/include" "${CMAKE_CURRENT_LIST_DIR}/generated")
target_compile_options(atlas_normalized_determinant_context PRIVATE -Wall -Wextra)
if(EMSCRIPTEN)
 set_target_properties(atlas_normalized_determinant_context PROPERTIES OUTPUT_NAME normalized-determinant-context SUFFIX ".mjs")
 target_link_options(atlas_normalized_determinant_context PRIVATE "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=node" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-sALLOW_MEMORY_GROWTH=1" "-fexceptions" "--no-entry")
 target_compile_options(atlas_normalized_determinant_context PRIVATE -fexceptions)
endif()
