cmake_minimum_required(VERSION 3.20)
project(AtlasControlTrace LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_control_trace EXCLUDE_FROM_ALL driver.cpp)
target_include_directories(atlas_control_trace PRIVATE frozen/cpp/include generated)
target_compile_options(atlas_control_trace PRIVATE -Wall -Wextra -Wno-unused-variable)
if(EMSCRIPTEN)
 target_compile_options(atlas_control_trace PRIVATE -fexceptions)
 set_target_properties(atlas_control_trace PROPERTIES OUTPUT_NAME control-trace SUFFIX ".js")
 target_link_options(atlas_control_trace PRIVATE -fexceptions "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,worker,node" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-sALLOW_MEMORY_GROWTH=1" "-sSTACK_SIZE=1048576" "-sFILESYSTEM=0" "--no-entry")
endif()
