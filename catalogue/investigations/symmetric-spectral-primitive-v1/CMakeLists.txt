cmake_minimum_required(VERSION 3.20)
project(AtlasSpectralStudy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_spectral_study driver.cpp)
target_include_directories(atlas_spectral_study PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_options(atlas_spectral_study PRIVATE -Wall -Wextra -Wno-misleading-indentation -ffp-contract=off)
if(EMSCRIPTEN)
  set_target_properties(atlas_spectral_study PROPERTIES OUTPUT_NAME "spectral-study" SUFFIX ".js")
  target_link_options(atlas_spectral_study PRIVATE "-sENVIRONMENT=node" "-sEXIT_RUNTIME=1" "-sSTACK_SIZE=1048576" "-sALLOW_MEMORY_GROWTH=1")
endif()
