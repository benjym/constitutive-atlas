cmake_minimum_required(VERSION 3.20)
project(AtlasExpressionContext LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(atlas_expression_context EXCLUDE_FROM_ALL probe.cpp)
target_include_directories(atlas_expression_context PRIVATE frozen/cpp/include generated)
target_compile_options(atlas_expression_context PRIVATE -Wall -Wextra -Wno-unused-variable -fno-fast-math -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(atlas_expression_context PRIVATE -fexceptions)
 set_target_properties(atlas_expression_context PROPERTIES OUTPUT_NAME atlas-expression-context SUFFIX ".js")
 target_link_options(atlas_expression_context PRIVATE -fexceptions "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=node" "-sEXPORTED_RUNTIME_METHODS=['ccall']" "-sALLOW_MEMORY_GROWTH=1" "-sSTACK_SIZE=1048576" "--no-entry")
endif()
